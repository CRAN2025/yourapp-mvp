Normalizer for legacy values

(Your DB has variants like banktransfer, localdelivery.) Normalize before writing and when reading.

// src/utils/normalizeOptionId.ts
const ALIASES: Record<string, OptionId> = {
  banktransfer: 'bank-transfer',
  'card-payment': 'card', cardpayment: 'card',
  mobile: 'mobilemoney', mobilemoney: 'mobilemoney',
  cryptocurrency: 'crypto', cryptocurrencies: 'crypto',

  'customer-pickup': 'pickup', customerpickup: 'pickup',
  homedelivery: 'home', localdelivery: 'local',
  nationalcourier: 'national', internationaldelivery: 'international',
};

export function normalizeOptionId(x: string): OptionId | null {
  const key = x.trim().toLowerCase().replace(/\s+/g,'-');
  return (ALIASES[key] ?? (key as OptionId)) in OPTION_INDEX ? (ALIASES[key] ?? (key as OptionId)) : null;
}

3) Storefront header counts + modal: read from public and render from constants

Don’t hard-code strings. Filter canonical lists by the public arrays and keep their order.

// src/storefront/useSellerProfile.ts
import { onValue, ref } from 'firebase/database';
export function useSellerPublicProfile(sellerId: string) {
  const [data, setData] = useState<{paymentMethods: OptionId[]; deliveryOptions: OptionId[]}>();
  useEffect(() => {
    const unsub = onValue(ref(db, `public/${sellerId}`), snap => setData(snap.val() ?? {paymentMethods:[], deliveryOptions:[]}));
    return () => unsub();
  }, [sellerId]);
  return data;
}

// Header counts
const profile = useSellerPublicProfile(sellerId);
const paymentCount  = profile?.paymentMethods?.length ?? 0;
const deliveryCount = profile?.deliveryOptions?.length ?? 0;

// Payment Methods modal (Storefront)
const selected = new Set(profile?.paymentMethods ?? []);
const list = PAYMENT_OPTIONS
  .filter(o => selected.has(o.id))
  .sort((a,b) => a.order - b.order);

return (
  <Modal title="Payment Methods">
    <ul className="space-y-2">
      {list.map(o => (
        <li key={o.id} className="flex items-center gap-3">
          <span aria-hidden>{o.emoji}</span>
          <span>{o.label}</span>
        </li>
      ))}
    </ul>
  </Modal>
);


…and the same for Delivery Options using DELIVERY_OPTIONS.

This ensures the Storefront shows only the seller’s enabled options, in the same order/labels/emojis as Settings. The top badges (e.g., “7 Payment Methods”, “6 Delivery Options”) will now match.
